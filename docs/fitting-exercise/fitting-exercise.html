Regression

# 1) Linear regression model
lm_spec <- linear_reg() %>%
  set_engine("lm")

# 2) Fit model
lm_dose_fit <- lm_spec %>%
  fit(Y ~ DOSE, data = train_df)

# 3) Predict
lm_dose_preds <- predict(lm_dose_fit, new_data = test_df) %>%
  bind_cols(test_df %>% select(Y))

# 4) RMSE and R-squared
metrics(lm_dose_preds, truth = Y, estimate = .pred) %>%
  filter(.metric %in% c("rmse", "rsq"))

Model fitting

# Specify logistic regression model
log_spec <- logistic_reg() %>%
  set_engine("glm")

# Fit logistic model using DOSE only
log_dose_fit <- log_spec %>%
  fit(SEX ~ DOSE, data = train_df)

# Predict classes
log_dose_class <- predict(log_dose_fit, new_data = test_df, type = "class")

# Predict probabilities
log_dose_prob <- predict(log_dose_fit, new_data = test_df, type = "prob")

# Combine results
log_dose_results <- bind_cols(
  test_df %>% select(SEX),
  log_dose_class,
  log_dose_prob
